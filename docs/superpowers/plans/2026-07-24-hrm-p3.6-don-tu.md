```ts
export const HE_SO_OT = { ngay_thuong: 1.5, ngay_nghi: 2.0, ngay_le: 3.0 } as const;
export type LoaiNgayOt = keyof typeof HE_SO_OT;

export function suyHeSoOt(input: {
  ngay: string;                       // YYYY-MM-DD
  laNgayLe: boolean;
  ngayLamViecTrongTuan?: number[];    // 0=CN … 6=T7; undefined/rỗng = chưa cấu hình
}): { loaiNgayOt: LoaiNgayOt; heSoOt: number };

export function tinhSoGioOt(gioTu: string, gioDen: string): number;

export function tinhSoNgayNghi(input: {
  tuNgay: string;
  denNgay: string;
  buoi?: string;                      // 'ca_ngay' | 'sang' | 'chieu'
  ngayLeTrongKhoang: string[];        // các ngày YYYY-MM-DD là ngày lễ
  ngayLamViecTrongTuan?: number[];
}): number;
```

**Ba điểm quyết định phải đúng:**

1. **Thứ tự ưu tiên hệ số**: lễ thắng ngày nghỉ. Lễ rơi vào Chủ nhật → `ngay_le` (3.0), không phải `ngay_nghi`. Kiểm `laNgayLe` **trước**.
2. **`ngayLamViecTrongTuan` rỗng/undefined = chưa cấu hình**, nghĩa là **không coi ngày nào là ngày nghỉ** — cùng quy ước đã ghi ở màn hồ sơ NV ("Bỏ trống nghĩa là chưa cấu hình — hệ thống sẽ không coi ngày nào là ngày nghỉ, không phải nghỉ tất cả các ngày"). Suy ngược lại là biến mọi đơn OT thành hệ số 2.0.
3. **`tinhSoGioOt` phải xử lý qua nửa đêm**: `gioTu` 22:00, `gioDen` 02:00 → 4 giờ, không phải âm.

`tinhSoNgayNghi`: đếm các ngày trong `[tuNgay, denNgay]` **không** phải ngày lễ và **có** trong `ngayLamViecTrongTuan` (nếu đã cấu hình). Đúng 1 ngày + `buoi` là `sang`/`chieu` → **0.5**.

```ts
import { describe, it, expect } from '@jest/globals';
import { suyHeSoOt, tinhSoGioOt, tinhSoNgayNghi, HE_SO_OT } from './luat-don';

const T2_DEN_T6 = [1, 2, 3, 4, 5];

describe('suyHeSoOt', () => {
  it('ngày làm việc bình thường → 1.5', () => {
    expect(suyHeSoOt({ ngay: '2026-07-22', laNgayLe: false, ngayLamViecTrongTuan: T2_DEN_T6 }))
      .toEqual({ loaiNgayOt: 'ngay_thuong', heSoOt: 1.5 });
  });

  it('ngày không thuộc lịch làm việc → 2.0', () => {
    // 2026-07-26 là Chủ nhật.
    expect(suyHeSoOt({ ngay: '2026-07-26', laNgayLe: false, ngayLamViecTrongTuan: T2_DEN_T6 }))
      .toEqual({ loaiNgayOt: 'ngay_nghi', heSoOt: 2.0 });
  });

  it('ngày lễ → 3.0', () => {
    expect(suyHeSoOt({ ngay: '2026-09-02', laNgayLe: true, ngayLamViecTrongTuan: T2_DEN_T6 }))
      .toEqual({ loaiNgayOt: 'ngay_le', heSoOt: 3.0 });
  });

  /**
   * Thứ tự ưu tiên là thứ dễ làm sai nhất: lễ rơi vào Chủ nhật phải tính LỄ.
   * Kiểm ngày nghỉ trước sẽ trả 2.0 và công ty trả thiếu tiền cho người đi làm
   * đúng ngày đáng được trả cao nhất.
   */
  it('lễ rơi vào Chủ nhật → tính LỄ, không phải ngày nghỉ', () => {
    expect(suyHeSoOt({ ngay: '2026-07-26', laNgayLe: true, ngayLamViecTrongTuan: T2_DEN_T6 }))
      .toEqual({ loaiNgayOt: 'ngay_le', heSoOt: 3.0 });
  });

  /**
   * Chưa cấu hình lịch làm việc KHÔNG có nghĩa là nghỉ tất cả các ngày — cùng
   * quy ước đã ghi ở màn hồ sơ nhân viên. Hiểu ngược lại thì mọi đơn OT của
   * người chưa được gán lịch đều thành hệ số 2.0.
   */
  it('chưa cấu hình lịch làm việc → coi là ngày thường', () => {
    expect(suyHeSoOt({ ngay: '2026-07-26', laNgayLe: false }))
      .toEqual({ loaiNgayOt: 'ngay_thuong', heSoOt: 1.5 });
    expect(suyHeSoOt({ ngay: '2026-07-26', laNgayLe: false, ngayLamViecTrongTuan: [] }))
      .toEqual({ loaiNgayOt: 'ngay_thuong', heSoOt: 1.5 });
  });

  it('HE_SO_OT là nguồn sự thật duy nhất của ba con số', () => {
    expect(HE_SO_OT).toEqual({ ngay_thuong: 1.5, ngay_nghi: 2.0, ngay_le: 3.0 });
  });
});

describe('tinhSoGioOt', () => {
  it('trong ngày', () => {
    expect(tinhSoGioOt('18:00', '20:30')).toBe(2.5);
  });

  /** Ca OT đêm là chuyện thường — trừ thẳng sẽ ra số âm. */
  it('qua nửa đêm → cộng 24h, không ra số âm', () => {
    expect(tinhSoGioOt('22:00', '02:00')).toBe(4);
  });

  it('bằng nhau → 0', () => {
    expect(tinhSoGioOt('18:00', '18:00')).toBe(0);
  });
});

describe('tinhSoNgayNghi', () => {
  it('một ngày làm việc → 1', () => {
    expect(tinhSoNgayNghi({
      tuNgay: '2026-07-22', denNgay: '2026-07-22',
      ngayLeTrongKhoang: [], ngayLamViecTrongTuan: T2_DEN_T6,
    })).toBe(1);
  });

  it('nửa buổi → 0.5', () => {
    expect(tinhSoNgayNghi({
      tuNgay: '2026-07-22', denNgay: '2026-07-22', buoi: 'sang',
      ngayLeTrongKhoang: [], ngayLamViecTrongTuan: T2_DEN_T6,
    })).toBe(0.5);
  });

  /** Nghỉ T6→T3 vắt qua T7+CN: chỉ tính 3 ngày làm việc. */
```

**Đây là task quan trọng nhất của phase.** Hiện `@UseGuards(JwtGuard)` ở cấp controller và không route ghi nào có `AdminGuard`, `create` đọc `employeeId` từ body → **bất kỳ ai cũng tạo/sửa/xoá đơn của người khác và tự duyệt đơn của mình**.

**Hai nhóm route:**

```
Tự phục vụ (JwtGuard, employeeId suy từ token):
  GET    /don-cham-cong/cua-toi        đơn của chính mình
  POST   /don-cham-cong/cua-toi        nộp đơn cho chính mình
  DELETE /don-cham-cong/cua-toi/:id    tự huỷ đơn CÒN cho_duyet của chính mình

Quản trị (thêm AdminGuard):
  GET /  ·  POST /  ·  PUT /:id  ·  DELETE /:id  ·  GET /:id
```

**Bẫy thứ tự route:** `cua-toi` phải khai **trước** `:id`, nếu không NestJS khớp `"cua-toi"` thành `:id`. Cùng lỗi đã được khoá bằng test ở `ban-ghi-cham-cong.controller.spec.ts` và `thiet-bi-cham-cong.controller.ts`.

**`TaoDonCuaToiDto` KHÔNG có `employeeId`** — không phải "bỏ qua nếu gửi" mà là **không tồn tại trường đó**, để `forbidNonWhitelisted` chặn ngay ở tầng pipe.

**Chặn tự duyệt** (§6.1 spec): trong `updateStatus`, nếu `trangThai` là `da_duyet`/`tu_choi` và `req.user.id` trùng `userId` của nhân viên đứng tên đơn → `ForbiddenException({ code: 'KHONG_TU_DUYET_DON', ... })`. Ghi `nguoiDuyetId` + `thoiDiemDuyet`.

**Test bắt buộc — đây là phần đáng giá nhất:**
